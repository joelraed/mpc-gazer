cmake_minimum_required(VERSION 3.15)
project(MPCGazerDSP VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mpcgazer_dsp STATIC src/GazerDSP.cpp)
set_target_properties(mpcgazer_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mpcgazer_dsp PUBLIC src)
add_library(MPCGazer SHARED src/VST2Entry.cpp)
target_link_libraries(MPCGazer PRIVATE mpcgazer_dsp)
set_target_properties(MPCGazer PROPERTIES PREFIX "" OUTPUT_NAME "MPCGazer")
add_executable(mpcgazer_smoke tests/smoke.cpp)
target_link_libraries(mpcgazer_smoke PRIVATE mpcgazer_dsp)
enable_testing()
add_test(NAME smoke COMMAND mpcgazer_smoke)

add_executable(mpcgazer_celestial_glow tests/celestial_glow.cpp)
target_link_libraries(mpcgazer_celestial_glow PRIVATE mpcgazer_dsp)
add_test(NAME celestial_glow COMMAND mpcgazer_celestial_glow)
